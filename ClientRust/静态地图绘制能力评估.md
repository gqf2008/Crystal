## 🎯 Rust 实现方案

### **方案 1: 添加瓦片绘制方法到 CellInfo**

```rust
// ClientRust/src/objects/map_code.rs

impl CellInfo {
    /// 绘制地表层 (Back Layer)
    /// 
    /// C# Reference: Client/MirScenes/GameScene.cs lines 11640-11662
    /// 
    /// 特点:
    /// - 只渲染偶数坐标 (x%2==0 && y%2==0) - 性能优化
    /// - BackImage 需要屏蔽高3位标记 (& 0x1FFFFFFF)
    /// - 使用 MapLibs[back_index] 图库
    pub fn draw_back_layer(
        &self,
        ctx: &mut Context,
        canvas: &mut Canvas,
        map_libs: &[MLibrary],
        draw_x: f32,
        draw_y: f32,
    ) -> GameResult {
        if self.back_image == 0 || self.back_index == -1 {
            return Ok(());
        }
        
        // 屏蔽高3位标记，索引从1开始需要减1
        let index = ((self.back_image & 0x1FFFFFFF) - 1) as usize;
        let lib_index = self.back_index as usize;
        
        if lib_index < map_libs.len() {
            map_libs[lib_index].draw(ctx, canvas, index, draw_x, draw_y, 
                                     ggez::graphics::Color::WHITE, false)?;
        }
        
        Ok(())
    }
    
    /// 绘制建筑层 (Middle Layer)
    /// 
    /// C# Reference: Client/MirScenes/GameScene.cs lines 11668-11696
    /// 
    /// 特点:
    /// - 渲染所有坐标（不限奇偶）
    /// - 尺寸验证：只渲染 48x32 或 96x64 瓦片
    /// - 向下多渲染5格 (ViewRangeY + 5) 避免高大建筑被裁剪
    pub fn draw_middle_layer(
        &self,
        ctx: &mut Context,
        canvas: &mut Canvas,
        map_libs: &[MLibrary],
        draw_x: f32,
        draw_y: f32,
        cell_width: i32,
        cell_height: i32,
    ) -> GameResult {
        let index = (self.middle_image - 1) as usize;
        
        if self.middle_image <= 0 || self.middle_index == -1 {
            return Ok(());
        }
        
        let lib_index = self.middle_index as usize;
        if lib_index >= map_libs.len() {
            return Ok(());
        }
        
        // 尺寸验证 (防止绘制条状错误瓦片)
        let size = map_libs[lib_index].get_size(index);
        if size.is_none() {
            return Ok(());
        }
        let (width, height) = size.unwrap();
        
        // 只允许单格 (48x32) 或双格 (96x64) 尺寸
        let valid_size = (width == cell_width && height == cell_height) ||
                         (width == cell_width * 2 && height == cell_height * 2);
        
        if !valid_size {
            return Ok(());
        }
        
        map_libs[lib_index].draw(ctx, canvas, index, draw_x, draw_y,
                                 ggez::graphics::Color::WHITE, false)?;
        
        Ok(())
    }
    
    /// 绘制前景层 (Front Layer)
    /// 
    /// C# Reference: Client/MirScenes/GameScene.cs lines 11702-11750
    /// 
    /// 特点:
    /// - FrontImage 需要屏蔽高位标记 (& 0x7FFF)
    /// - 支持门动画 (DoorIndex > 0)
    /// - 尺寸验证同 Middle Layer
    pub fn draw_front_layer(
        &self,
        ctx: &mut Context,
        canvas: &mut Canvas,
        map_libs: &[MLibrary],
        draw_x: f32,
        draw_y: f32,
        cell_width: i32,
        cell_height: i32,
        door_offset: Option<i32>, // 门动画偏移（如果有门）
    ) -> GameResult {
        // 屏蔽高位标记，索引从1开始需要减1
        let mut index = ((self.front_image & 0x7FFF) - 1) as i32;
        
        if index == -1 || self.front_index == -1 {
            return Ok(());
        }
        
        let file_index = self.front_index as usize;
        if file_index >= map_libs.len() || file_index == 200 {
            // 修复旧版 4.map 的随机坏点
            return Ok(());
        }
        
        // 门动画处理
        if let Some(offset) = door_offset {
            if self.door_index > 0 {
                // 门打开动画：index += (doorFrame + 1) * DoorOffset
                index += (offset + 1) * self.door_offset as i32;
            }
        }
        
        // 尺寸验证
        let size = map_libs[file_index].get_size(index as usize);
        if size.is_none() {
            return Ok(());
        }
        let (width, height) = size.unwrap();
        
        let valid_size = (width == cell_width && height == cell_height) ||
                         (width == cell_width * 2 && height == cell_height * 2);
        
        if index < 0 || !valid_size {
            return Ok(());
        }
        
        map_libs[file_index].draw(ctx, canvas, index as usize, draw_x, draw_y,
                                  ggez::graphics::Color::WHITE, false)?;
        
        Ok(())
    }
    
    /// 绘制瓦片动画层 (Tile Animation Layer)
    /// 
    /// C# Reference: Client/MirScenes/GameScene.cs lines 11862-11876
    /// 
    /// 特点:
    /// - 固定使用 MapLibs[190] (Shanda 动画库)
    /// - 动画偏移异或 0x2000 (offset ^ 0x2000)
    /// - 循环动画：index += offset * (frame_count % total_frames)
    /// - 使用 DrawUp (从底部向上绘制)
    pub fn draw_tile_animation(
        &self,
        ctx: &mut Context,
        canvas: &mut Canvas,
        map_libs: &[MLibrary],
        draw_x: f32,
        draw_y: f32,
        animation_count: u32, // 全局动画计数器
    ) -> GameResult {
        let index = self.tile_animation_image;
        let frames = self.tile_animation_frames;
        
        if index <= 0 || frames == 0 {
            return Ok(());
        }
        
        // 动画库固定为 MapLibs[190]
        const ANIMATION_LIB: usize = 190;
        if ANIMATION_LIB >= map_libs.len() {
            return Ok(());
        }
        
        // 索引从1开始，需要减1
        let mut index = (index - 1) as i32;
        
        // 动画偏移异或 0x2000
        let offset = (self.tile_animation_offset ^ 0x2000) as i32;
        
        // 循环动画计算
        index += offset * ((animation_count % frames as u32) as i32);
        
        if index < 0 {
            return Ok(());
        }
        
        // 使用 draw_up (从底部向上绘制)
        map_libs[ANIMATION_LIB].draw_up(ctx, canvas, index as usize, draw_x, draw_y)?;
        
        Ok(())
    }
}
```

### **方案 2: MapReader 添加绘制方法**

```rust
impl MapReader {
    /// 绘制整个地图视野范围
    /// 
    /// 参数:
    /// - view_center: 视野中心坐标 (通常是玩家位置)
    /// - view_range: 视野半径 (X, Y)
    /// - offset: 屏幕偏移 (用于平滑滚动)
    /// - map_libs: 地图图库数组 (400个)
    /// - animation_count: 当前动画帧计数
    pub fn draw_map(
        &self,
        ctx: &mut Context,
        canvas: &mut Canvas,
        view_center: Point,
        view_range: (i32, i32),
        offset: Point,
        map_libs: &[MLibrary],
        animation_count: u32,
    ) -> GameResult {
        const CELL_WIDTH: i32 = 48;
        const CELL_HEIGHT: i32 = 32;
        
        let (view_range_x, view_range_y) = view_range;
        
        // ========== 阶段1: 绘制地表层 (只绘制偶数格) ==========
        for y in (view_center.y - view_range_y)..=(view_center.y + view_range_y) {
            if y <= 0 || y % 2 == 1 || y >= self.height {
                continue;
            }
            
            let draw_y = ((y - view_center.y + offset.y) * CELL_HEIGHT + offset.y) as f32;
            
            for x in (view_center.x - view_range_x)..=(view_center.x + view_range_x) {
                if x <= 0 || x % 2 == 1 || x >= self.width {
                    continue;
                }
                
                let draw_x = ((x - view_center.x + offset.x) * CELL_WIDTH - offset.x + offset.x) as f32;
                
                if let Some(cell) = self.get_cell(x, y) {
                    cell.draw_back_layer(ctx, canvas, map_libs, draw_x, draw_y)?;
                }
            }
        }
        
        // ========== 阶段2: 绘制建筑层 (渲染所有格子) ==========
        for y in (view_center.y - view_range_y)..=(view_center.y + view_range_y + 5) {
            if y <= 0 || y >= self.height {
                continue;
            }
            
            let draw_y = ((y - view_center.y + offset.y) * CELL_HEIGHT + offset.y) as f32;
            
            for x in (view_center.x - view_range_x)..=(view_center.x + view_range_x) {
                if x < 0 || x >= self.width {
                    continue;
                }
                
                let draw_x = ((x - view_center.x + offset.x) * CELL_WIDTH - offset.x + offset.x) as f32;
                
                if let Some(cell) = self.get_cell(x, y) {
                    cell.draw_middle_layer(ctx, canvas, map_libs, draw_x, draw_y, 
                                          CELL_WIDTH, CELL_HEIGHT)?;
                }
            }
        }
        
        // ========== 阶段3: 绘制前景层 ==========
        for y in (view_center.y - view_range_y)..=(view_center.y + view_range_y + 5) {
            if y <= 0 || y >= self.height {
                continue;
            }
            
            let draw_y = ((y - view_center.y + offset.y) * CELL_HEIGHT + offset.y) as f32;
            
            for x in (view_center.x - view_range_x)..=(view_center.x + view_range_x) {
                if x < 0 || x >= self.width {
                    continue;
                }
                
                let draw_x = ((x - view_center.x + offset.x) * CELL_WIDTH - offset.x + offset.x) as f32;
                
                if let Some(cell) = self.get_cell(x, y) {
                    cell.draw_front_layer(ctx, canvas, map_libs, draw_x, draw_y,
                                         CELL_WIDTH, CELL_HEIGHT, None)?;
                }
            }
        }
        
        Ok(())
    }
}
```

```rust
use ggez::{Context, GameResult, graphics::{Canvas, Color}};
use client_rust::graphics::mlibrary::MLibrary;
use client_rust::objects::map_code::MapReader;

struct MapViewer {
    map: MapReader,
    map_libs: Vec<MLibrary>,
}

impl MapViewer {
    fn new() -> GameResult<Self> {
        let map = MapReader::new("d:/Mir2/Map/0.map")?;
        
        // 加载必需的地图图库 (0-9号库)
        let mut map_libs = Vec::new();
        for i in 0..10 {
            let path = format!("d:/Mir2/Data/Tiles{}", i);
            map_libs.push(MLibrary::new(&path));
        }
        
        Ok(Self { map, map_libs })
    }
    
    fn draw(&mut self, ctx: &mut Context, canvas: &mut Canvas) -> GameResult {
        // 绘制 10x10 区域的地图
        for y in 100..110 {
            for x in 100..110 {
                if let Some(cell) = self.map.get_cell(x, y) {
                    let draw_x = ((x - 100) * 48) as f32;
                    let draw_y = ((y - 100) * 32) as f32;
                    
                    // 只绘制地表层（最简单）
                    if cell.back_image > 0 && cell.back_index >= 0 {
                        let index = ((cell.back_image & 0x1FFFFFFF) - 1) as usize;
                        let lib_index = cell.back_index as usize;
                        
                        if lib_index < self.map_libs.len() {
                            self.map_libs[lib_index].draw(
                                ctx, canvas, index, draw_x, draw_y, Color::WHITE, false
                            )?;
                        }
                    }
                }
            }
        }
        
        Ok(())
    }
}

fn main() -> GameResult {
    let cb = ggez::ContextBuilder::new("map_viewer", "test");
    let (mut ctx, event_loop) = cb.build()?;
    let mut viewer = MapViewer::new()?;
    
    ggez::event::run(ctx, event_loop, viewer)
}
```
