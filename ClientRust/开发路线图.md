### **任务 2: 实现键盘移动** (HIGH)

**目标**: 按WASD/方向键移动玩家

**实现**:
```rust
// 在 game_scene.rs 中添加:
fn handle_key_input(&mut self, key: KeyCode, pressed: bool) {
    if !pressed { return; }
    
    if let Some(ref user) = self.user {
        let current_pos = user.player.map_object.location();
        let new_pos = match key {
            KeyCode::W | KeyCode::Up    => Point::new(current_pos.x, current_pos.y - 2),
            KeyCode::S | KeyCode::Down  => Point::new(current_pos.x, current_pos.y + 2),
            KeyCode::A | KeyCode::Left  => Point::new(current_pos.x - 2, current_pos.y),
            KeyCode::D | KeyCode::Right => Point::new(current_pos.x + 2, current_pos.y),
            _ => return,
        };
        
        // 检查地图边界
        if let Some(ref map) = self.map_control {
            if new_pos.x >= 2 && new_pos.x < map.width - 2 &&
               new_pos.y >= 2 && new_pos.y < map.height - 2 {
                // 移动玩家
                self.move_object(user.player.map_object.object_id(), new_pos);
                self.player_x = new_pos.x;
                self.player_y = new_pos.y;
                
                tracing::info!("Player moved to ({}, {})", new_pos.x, new_pos.y);
            }
        }
    }
}
```

fn draw(&self, ctx: &mut Context, canvas: &mut Canvas, draw_location: Point) -> GameResult {

    let action = self.player.map_object.current_action;

    let frame = self.player.map_object.frame_index;

        (MirGender::Female, MirClass::Warrior) => "Data/Hum2.Lib",
