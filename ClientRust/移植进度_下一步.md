```csharp
{

    try
    {
        byte[] sum;
        using (MD5 md5 = MD5.Create())
```

```rust
GameEvent::Connected => {
    self.record_status("Connected to server successfully");
    self.connecting = false;
    
    // 发送客户端版本验证
    if let Some(client) = &self.game_client {
        use mir2_shared::packets::client::ClientVersion;
        use std::fs::File;
        use std::io::Read;
        use md5::{Md5, Digest};
        
        // 计算可执行文件的 MD5
        let exe_path = std::env::current_exe().ok();
        let version_hash = if let Some(path) = exe_path {
            if let Ok(mut file) = File::open(path) {
                let mut hasher = Md5::new();
                let mut buffer = Vec::new();
                if file.read_to_end(&mut buffer).is_ok() {
                    hasher.update(&buffer);
                    hasher.finalize().to_vec()
                } else {
                    vec![0u8; 16]  // 空哈希
                }
            } else {
                vec![0u8; 16]
            }
        } else {
            vec![0u8; 16]
        };
        
        let version_packet = ClientVersion {
            version_hash,
        };
        
        if let Ok(mut client) = client.try_write() {
            client.send_packet(version_packet);
            self.record_status("Sending client version...");
        }
    }
}
```

```rust
fn submit_new_account(&mut self, registration: AccountRegistration) {
    if let Some(client) = &self.game_client {
        use mir2_shared::packets::client::NewAccount;
        
        let packet = NewAccount {
            account_id: registration.account_id,
            password: registration.password,
            email_address: registration.email,
            user_name: registration.username,
            secret_question: registration.secret_question,
            secret_answer: registration.secret_answer,
```
