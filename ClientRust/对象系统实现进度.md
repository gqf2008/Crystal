```rust
impl MapObject {
    pub fn add_buff(&mut self, buff: BuffType) {
        if !self.buffs.contains(&buff) {
            self.buffs.push(buff);
            tracing::debug!("Added buff {:?} to object {}", buff, self.object_id);
        }
    }
    
    pub fn remove_buff(&mut self, buff: BuffType) {
        if let Some(pos) = self.buffs.iter().position(|b| *b == buff) {
            self.buffs.remove(pos);
            tracing::debug!("Removed buff {:?} from object {}", buff, self.object_id);
        }
    }
    
    pub fn clear_buffs(&mut self) {
        if !self.buffs.is_empty() {
            self.buffs.clear();
            tracing::debug!("Cleared all buffs from object {}", self.object_id);
        }
    }
}
```

fn draw(&self, ctx: &mut Context, canvas: &mut Canvas, draw_location: Point) -> GameResult {
    // 1. 获取怪物纹理
    let texture = get_monster_texture(self.base_image, self.action, self.direction)?;
    
    // 2. 计算帧索引
    let frame = self.animation_count / self.frame_interval;

    draw_effects(canvas, draw_location, &self.effects);
    
    Ok(())
}

#### 4. 处理对象移动和插值
```rust
impl MapObject {
    pub fn update_movement(&mut self, delta_time: f32) {
        if self.is_moving() {
            // 平滑移动插值
            let progress = self.movement_progress + delta_time * MOVEMENT_SPEED;
            
            if progress >= 1.0 {
                // 移动完成
                self.current_location = self.map_location;
                self.movement_progress = 0.0;
            } else {
                // 插值计算当前位置
                self.movement_progress = progress;
                self.current_location = lerp(
                    self.movement_start,
                    self.map_location,
                    progress
                );
            }
            
            // 更新绘制位置
            self.update_draw_location();
        }
    }
}
```
