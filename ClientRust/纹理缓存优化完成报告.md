## 🔧 技术实现

### 1. MLibrary 纹理缓存

**文件**: `src/graphics/mlibrary.rs`

**添加字段**:
```rust
pub struct MLibrary {
    // ... 现有字段 ...
    
    // ggez Image 缓存 (对应 C# DXManager.TextureList)
    ggez_texture_cache: HashMap<usize, ggez::graphics::Image>,
    
    // 缓存访问时间 (用于 LRU 清理)
    ggez_cache_access_time: HashMap<usize, std::time::Instant>,
}
```

**核心方法**:
```rust
/// 获取或创建缓存的 ggez 纹理
/// 
/// 对应 C# MImage.CreateTexture() + DXManager.TextureList
pub fn get_or_create_texture(
    &mut self,
    ctx: &mut ggez::Context,
    index: usize,
) -> io::Result<&ggez::graphics::Image> {
    if !self.ggez_texture_cache.contains_key(&index) {
        // 缓存未命中 - 加载并创建纹理
        let (info, rgba_data) = self.load_rgba_data(index)?;
        let image = Image::from_pixels(ctx, &rgba_data, ...);
        self.ggez_texture_cache.insert(index, image);
    }
    
    // 更新访问时间 (LRU)
    self.ggez_cache_access_time.insert(index, Instant::now());
    
    // 返回缓存的纹理
    Ok(self.ggez_texture_cache.get(&index).unwrap())
}

/// 清理长时间未使用的纹理缓存
/// 
/// 对应 C# DXManager.CleanUp()
pub fn cleanup_old_textures(&mut self, max_age: Duration) {
    let now = Instant::now();
    
    self.ggez_texture_cache.retain(|&idx, _| {
        if let Some(access_time) = self.ggez_cache_access_time.get(&idx) {
            now.duration_since(*access_time) <= max_age
        } else {
            false
        }
    });
}
```

### 3. 自动清理机制

**文件**: `src/scenes/game_scene.rs`

**定期清理**:
```rust
pub fn draw(&mut self, ctx: &mut Context, canvas: &mut Canvas) -> GameResult<()> {
    // 每 5 分钟清理一次超过 10 分钟未使用的纹理
    static mut LAST_CLEANUP_TIME: Option<Instant> = None;
    unsafe {
        let now = Instant::now();
        if LAST_CLEANUP_TIME.is_none() || 
           now.duration_since(LAST_CLEANUP_TIME.unwrap()) > Duration::from_secs(300) {
            self.cleanup_texture_cache();
            LAST_CLEANUP_TIME = Some(now);
        }
    }
    
    // ... 渲染逻辑 ...
}

fn cleanup_texture_cache(&mut self) {
    let max_age = Duration::from_secs(600); // 10 分钟
    let libs = get_all_map_libraries();
    
    for lib in libs {
        lib.lock().unwrap().cleanup_old_textures(max_age);
    }
}
```
