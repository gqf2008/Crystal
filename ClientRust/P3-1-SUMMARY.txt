## 🎯 使用示例

```rust
// 1. 初始化 (app.rs - 已完成)
let mut character_renderer = CharacterRenderer::new();
character_renderer.load_chrsel_library("Data/ChrSel.lib")?;

// 2. 加载精灵 (select_scene.rs - 已集成)
let (_, color_image) = character_renderer
    .load_character_color_image(
        MirClass::Warrior,
        MirGender::Male,
        0
    )?;

// 3. 显示 (egui)
let texture = ui.ctx().load_texture("preview", color_image, ...);
ui.image(&texture);
```
